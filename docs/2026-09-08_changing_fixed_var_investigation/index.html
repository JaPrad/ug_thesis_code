Obtaining Duplicates

The fixed columns along with id include state, district, caste_category and religion.

We will trim both datasets to these columns, and merge them in long format.

monthly_panel_pre_covid |>
  pivot_longer(
    cols = matches("_\\d{4}_\\d{2}$"),
    names_to = c(".value", "month_id"),
    names_pattern = "^(.*)_(\\d{4}_\\d{2})$"
  ) |>
  filter(!is.na(month_slot)) |> # remove observation when not part of sample
  select(
    id,
    state,
    district,
    caste_category,
    religion,
    month_id
  ) |>
  add_row(
    monthly_panel_post_covid |>
      pivot_longer(
        cols = matches("_\\d{4}_\\d{2}$"),
        names_to = c(".value", "month_id"),
        names_pattern = "^(.*)_(\\d{4}_\\d{2})$"
      ) |>
      filter(!is.na(month_slot)) |>
      select(
        id,
        state,
        district,
        caste_category,
        religion,
        month_id
      )
  ) -> merged_trimmed_data

Testing Data Wrangling Errors

We will check if the stated results match with raw data, in order to rule out errors due to bugs in code used to create the panel.

We define a function first.

check_raw <- function(id_arg, month_id_arg) {
  duplicates |>
    filter(id == id_arg, month_id == month_id_arg) |>
    select(-month_id_pasted) |>
    mutate(source = "Panel", .before = 1) -> panel_row

  read_csv(
    here(
      "data",
      "cmie",
      "cmie_income_raw",
      "member",
      cphs_monthid_to_income_filename(month_id_arg, "member")
    ),
    show_col_types = FALSE
  ) |>
    clean_names() |>
    mutate(id = paste0(hh_id, "_", mem_id)) |>
    filter(id == id_arg) |>
    select(id, state, district, caste_category, religion) |>
    mutate(source = "Raw", .before = 1) -> raw_row

  panel_row |>
    add_row(raw_row) %>%
    return()
}

get_count_matrix <- function(df, col_name, num_sample) {
  df |>
    rename(current_col = !!sym(col_name)) |>
    select(id, current_col) -> df_temp # Rename and select current column with id

  df_temp |>
    distinct(current_col) |>
    pull() -> distinct_classes # All Distinct classes

  class_ids <- list() # Empty list to store count

  # Obtain count:
  for (class_curr in distinct_classes) {
    class_ids[[class_curr]] <- df_temp |>
      filter(current_col == class_curr) |>
      distinct(id) |>
      pull(id)
  }

  expand_grid(distinct_classes, distinct_classes) |>
    rowwise() |>
    mutate(
      n = length(intersect(
        class_ids[[distinct_classes...1]],
        class_ids[[distinct_classes...2]]
      ))
    ) |>
    ungroup() |>
    mutate(percentage = glue("{round((n/num_sample) * 100, 3)} %")) |>
    mutate(across(n, comma)) |>
    mutate(n = glue("{percentage} <br> ({n})")) |>
    select(-percentage) |>
    pivot_wider(names_from = distinct_classes...2, values_from = n) |>
    rename(Class = distinct_classes...1) %>%

    {
      color_cols <- grep("_color$", names(.), value = TRUE)
      value_cols <- sub("_color$", "", color_cols)

      datatable(
        .,
        escape = FALSE,
        rownames = FALSE,
        extensions = c("FixedHeader", "FixedColumns"),

        options = list(
          pageLength = 20,
          scrollX = TRUE,
          autoWidth = TRUE,
          fixedHeader = TRUE,
          fixedColumns = list(leftColumns = 1),
          searching = TRUE, # Disables the global search/filter box
          paging = FALSE, # Disables pagination (shows all rows at once)
          ordering = TRUE, # Disables column sorting click interactions
          info = TRUE, # Disables the "Showing 1 to N of N entries" text

          columnDefs = list(
            list(
              visible = FALSE,
              targets = which(names(.) %in% color_cols) - 1
            )
          )
        ),

        callback = JS(
          "
        table.on('mouseenter', 'tbody td', function() {

          var idx = table.cell(this).index();

          if (!idx) return;

          // Remove old hover highlighting
          $(table.cells().nodes()).removeClass(
            'row-hover col-hover active-cell'
          );

          $(table.columns().header()).removeClass('col-hover');

          // Highlight entire row
          $(table.row(idx.row).nodes())
            .find('td')
            .addClass('row-hover');

          // Highlight entire column
          $(table.column(idx.column).nodes())
            .addClass('col-hover');

          // Highlight column header
          $(table.column(idx.column).header())
            .addClass('col-hover');

          // Stronger highlight for actual hovered cell
          $(this).addClass('active-cell');
        });

        table.on('mouseleave', 'tbody', function() {

          $(table.cells().nodes()).removeClass(
            'row-hover col-hover active-cell'
          );

          $(table.columns().header()).removeClass('col-hover');
        });
      "
        )
      ) %>%

        formatStyle(
          columns = 1,
          fontWeight = "bold"
        ) %>%

        formatStyle(
          columns = value_cols,
          valueColumns = color_cols,
          backgroundColor = JS("value")
        ) %>%

        htmlwidgets::prependContent(
          htmltools::tags$style(
            htmltools::HTML(
              "

            /* Remove DataTables zebra/default row backgrounds */
            table.dataTable tbody tr,
            table.dataTable tbody tr.odd,
            table.dataTable tbody tr.even {
              background-color: transparent !important;
            }

            /* Hovered row */
            table.dataTable td.row-hover {
              box-shadow:
                inset 0 0 0 9999px rgba(128,128,128,0.12) !important;
            }

            /* Hovered column */
            table.dataTable td.col-hover {
              box-shadow:
                inset 0 0 0 9999px rgba(128,128,128,0.12) !important;
            }

            /* Intersection / actual hovered cell */
            table.dataTable td.active-cell {
              box-shadow:
                inset 0 0 0 9999px rgba(128,128,128,0.25) !important;
            }

            /* Column heading */
            table.dataTable th.col-hover {
              box-shadow:
                inset 0 0 0 9999px rgba(128,128,128,0.20) !important;
            }

          "
            )
          )
        )
    }
}
